#+begin_src snippet :tangle "~/.emacs.d/snippets/go-ts-mode/middleware" :mkdirp yes

  # name: Middleware
  # key: _mw
  # --
  func $1(next http.Handler) http.Handler {
      return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
          $0

  	next.ServeHTTP(w, r)
      })
  }
